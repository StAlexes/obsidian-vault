| Взаимосвязь API и макроязыка |

В данном разделе показано, как преобразовать ReferenceObject из API в объект макроязыка и как выполнить обратное приведение — из объекта макроязыка в ReferenceObject

![Изображение](../icons/SectionExpanded.png)Пример приведения объекта макроязыка к ReferenceObject

C#

[Копировать](# "Копировать")

```
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Macros.ObjectModel;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.References.Users;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        Объект пользователь = НайтиОбъект("Группы и пользователи", "ID", 100);

        // Можем привести объект макроязыка к базовому классу, описывающему объект справочника (ReferenceObject)
        ReferenceObject referenceObject = (ReferenceObject)пользователь;

        // Можем привести к конкретному классу и воспользоваться его возможностями
        if (пользователь?.Тип == "Пользователь")
        {
            User user = (User)пользователь;
            string firstName = user.FirstName;
        }
    }
}
```

![Изображение](../icons/SectionExpanded.png)Пример получения объекта макроязыка из ReferenceObject

C#

[Копировать](# "Копировать")

```
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Macros.ObjectModel;
using TFlex.DOCs.Model.References;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        ReferenceObject foundObject = Context.Reference.Find(1); // Получаем объект при помощи API

        if (foundObject is null)
            return;

        Объект найденныйОбъект = Объект.CreateInstance(foundObject, Context); // Преобразуем ReferenceObject в Объект макроязыка
        ПоказатьДиалогСвойств(найденныйОбъект); // Используем возможности макроязыка
    }
}
```
