![Изображение](../icons/SectionExpanded.png)Пример создания напоминания

Напоминание может быть связано с любым объектом справочника, сообщением или заданием.

C#

[Копировать](# "Копировать")

```
using System;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.Mail.Reminders;
using TFlex.DOCs.Model.References.Documents;

public class CreationReminderMacro : MacroProvider
{
    public CreationReminderMacro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        // Находим объект с которым будет связано напоминание
        var documentReference = new DocumentReference(Context.Connection);
        var document = documentReference.Find(1);
        if (document == null)
            return;

        // Создаем напоминание по объекту.
        // Напоминание по элементу почты создается по аналогии: var reminder = new MailItemReminder(mailItem)
        var reminder = new ReferenceObjectReminder(document)
        {
            Date = DateTime.Now.AddMonths(6), // Задаем дату и время
            Text = String.Format("Напоминание по объекту '{0}'", document) // Задаем текст напоминания
        };

        reminder.Save(); // Сохраняем напоминание
    }
}
```
