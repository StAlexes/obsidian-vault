```
using System.Threading;
using TFlex.DOCs.Client;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Nomenclature;
using TFlex.DOCs.UI.Client.Common.ViewModels.MDI.Descriptions;

public class Macro : MacroProvider
{
    public Macro(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        var nomenclatureReference = new NomenclatureReference(Context.Connection); // Создаем экземпляр справочника "Электронная структура изделий"
        var nomenclatureObject = nomenclatureReference.Find(1); // Находим объект справочника

        var description = new MdiReferenceDescription // Создаем описание окна справочника
        {
            ReferenceInfo = nomenclatureReference.ParameterGroup.ReferenceInfo, // Указываем справочник
        };

        // Если задать корневой объект, то откроется вкладка для которой данный объект будет задан в качестве корневого. Т.е. программное выполнение команды "Открыть в новом окне"
        //description.RootObject = nomenclatureObject;

        Context.RunOnUIThread(() => ApplicationManager.OpenView(description, Context.Connection, CancellationToken.None)); // Открываем справочник в основном UI-потоке
    }
}
```

```
/*

TFlex.DOCs.IoC.Infrastructure.dll
TFlex.DOCs.UI.Client.dll
TFlex.DOCs.UI.Client.Common.dll*/

using System;
using TFlex.DOCs.Client;
using TFlex.DOCs.Model.Configuration.WorkingPages;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.UI.Client.Common.ViewModels;
using TFlex.DOCs.UI.Client.Common.ViewModels.Base;
using TFlex.DOCs.UI.Client.Common.ViewModels.WorkingPages;

public class Macro : MacroProvider
{
    public Macro(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        if (Context is not UIMacroContext uiMacroContext) // Проверяем, выполняется ли макрос в графическом интерфейсе пользователя
            return;

        WorkingPage page = WorkingPageManager.Find(Context.Connection, new Guid("Guid страницы")); // Находим рабочую страницу по ее Guid'у

        if (page is null || !page.CanUseInCurrentConfiguration()) // Если страница не найдена или не может быть использована в текущей конфигурации, то выходим
            return;

        var workingPageVm = new WorkingPageViewModel(page, uiMacroContext.OwnerViewModel as LayoutViewModel); // Создаем модель представления рабочей страницы
        Context.RunOnUIThread(() => ApplicationManager.OpenView(workingPageVm)); // Открываем страницу в основном UI-потоке
    }
}
```

```
/*

using System.Threading;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.References.ProjectManagement;
using TFlex.DOCs.UI.Client.Common.ViewModels.MDI.Descriptions;
using TFlex.DOCs.UI.Client.Common;

public class Macro(MacroContext context) : MacroProvider(context)
{
    public override void Run()
    {
        if (Context.ReferenceObject is not Project project) // Для примера открываем окно редактора проектов для текущего объекта
            return;

        var description = new MdiProjectEditorDescription // Создаем описание окна редактора проектов
        {
            ProjectIds = [project.Id]  // Передаем Id проекта
        };

        Context.RunOnUIThread(() => Services.MdiService.OpenView(description, Context.Connection, CancellationToken.None)); // Открываем редактор проектов в основном UI-потоке
    }
}
```
