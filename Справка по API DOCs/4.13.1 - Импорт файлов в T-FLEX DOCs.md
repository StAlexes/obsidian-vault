| Импорт файлов в T-FLEX DOCs |

API T-FLEX DOCs позволяет импортировать любые файлы из файловой системы в файловое хранилище T-FLEX DOCs. Ниже приводится пример реализации такой возможности.

![Изображение](../icons/SectionExpanded.png)Импорт файлов - общая концепция

Для организации импорта файлов предусмотрена функция [Import](9AB6D17F.htm)
класса [FileReference](7E06EE0D.htm)

C#

[Копировать](# "Копировать")

```
public FolderObject Import(string sourceFolder, ImportParameters parameters)
```

Где sourceFolder - исходная папка в файловой системе, parameters - параметры импорта. Функция возвращает папку файлового хранилища с импортированными объектами.

![Изображение](../icons/SectionExpanded.png)Импорт файлов - пример в виде макроса

Ниже приводится пример импорта файлов с использованием описанной выше функции [Import](9AB6D17F.htm):

C#

[Копировать](# "Копировать")

```
using System;
using System.Collections.Generic;
using System.Text;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Files;

public class ImportingFilesIntoDOCsMacro : MacroProvider
{
    // Исходная папка в файловой системе - из нее будут копироваться файлы
    private const string FolderPath = @"C:\!TEMP\FilesToImport";

    public ImportingFilesIntoDOCsMacro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        try
        {
            // Создаем экземпляр файлового справочника
            var fileReference = new FileReference(Context.Connection);

            // Задаем целевую папку для импорта файлов
            var parentFolder = fileReference.FindByRelativePath("Служебные файлы") as FolderObject;

            // Вызываем функцию импорта файлов, и получение объекта с описание содержимого в папке
            FolderObject result = fileReference.Import(FolderPath,
                new ImportParameters
                {
                    // Задаем папку назначения
                    DestinationFolder = parentFolder,
                    // Задаем значение, указывающее, требуется ли выполнять операцию рекурсивно для всех входящих папок
                    Recursive = true,
                    // Задаем значение, указывающее, требуется ли создавать новые типы
                    CreateClasses = true,
                    // Задаем значение, указывающее, требуется ли автоматическое применение изменений
                    AutoCheckIn = false
                }
            );

            // Выводим результат операции
            ViewResult(result);
        }
        catch (Exception e)
        {
            // Выводим сообщение об ошибке
            Сообщение("Ошибка", e.Message);
        }
    }

    // Выводим содержимое папки
    private void ViewResult(FolderObject folder)
    {
        // Создаем стек для обхода дерева объекта FolderObject
        var stack = new Stack<FolderObject>();

        // Добавляем корневой элемент
        stack.Push(folder);

        var sb = new StringBuilder();

        // Рекурсивно обходим содержимое папки
        while (stack.Count > 0)
        {
            FolderObject folderItem = stack.Pop();

            foreach (FileReferenceObject refObj in folderItem.Children)
            {
                sb.AppendLine(refObj.Path);

                if (refObj.IsFolder)
                    stack.Push((FolderObject)refObj);
            }
        }
        // Выводим сообщение со списком импортированных фалов
        Сообщение("Список файлов", sb.ToString());
    }
}
```

Результатом выполнения данного кода будет импорт всех файлов из директории "C:\!TEMP\FilesToImport" (включая подпапки) в папку "Служебные файлы" справочника "Файлы".
