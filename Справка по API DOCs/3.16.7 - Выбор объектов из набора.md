| Выбор объектов из набора |

Данный диалог позволяет осуществить выбор объектов из заданного набора объектов

![Изображение](../icons/SectionExpanded.png)Пример работы с диалогом выбора объектов из набора

[СоздатьДиалогВыбораОбъектовИзНабора(ОбъектыДляВыбора)](1DF6EB3A.htm)
- создает диалог выбора объектов из набора

C#

[Копировать](# "Копировать")

```
Объекты наборОбъектов = НайтиОбъекты("Группы и пользователи", "[Тип] Порождён от 'Сотрудник'"); // Формируем набор объектов
ДиалогВыбораОбъектовИзНабора диалог = СоздатьДиалогВыбораОбъектовИзНабора(наборОбъектов); // Создаем диалог выбора объектов из набора объектов

диалог.Заголовок = "Выбора объектов из набора"; // Устанавливаем заголовок окна
диалог.СкрытьСистемныеКолонки = true; // Скрываем системные колонки
диалог.ОтображатьСтрокуПоиска = true; // Отображаем строку поиска
диалог.ОтзеркалитьДиалог = true; // Включаем зеркальное отображение диалога
диалог.СортироватьВручную = true; // Включаем отображение кнопок для ручной сортировки

диалог.ДобавитьКолонку("Id", объект => объект.Идентификатор.ToString()); // Добавляем колонку ID
диалог.ДобавитьКолонку("Наименование", объект => объект.ToString());  // Добавляем колонку с отображением параметра по умолчанию

if (диалог.Показать()) // Отображаем диалог
{
    Объекты изначальыйНаборОбъектов = диалог.НаборОбъектов; // Получаем коллекцию из которой выбирали объекты
    Объекты выбранныеОбъекты = диалог.ВыбранныеОбъекты; // Получаем коллекцию выбранных объектов
}
```
