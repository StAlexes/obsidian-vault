| Экспорт grb в другой формат |

В данном примере демонстрируется возможность экспорта cad-документов (\*.grb) в другие форматы.

Поддерживаемые форматы:

- Файлы PDF ("pdf")
- Растровые ("bmp", "jpg", "gif", "tiff", "png", "tga")
- Метафайлы ("bmf", "emf", "wmf")
- Файлы AutoCAD ("dwg", "dxb", "dxf")
- Файлы IGES 3D ("igs", "iges")
- Файлы Parasolid ("xmt\_bin", "xmt\_txt", "x\_t", "x\_b")
- Файлы Inventor ("iv")
- Файлы STEP AP203/AP214 3D ("stp", "step")
- Файлы "3dm", "x3d", "stl", "wrl"

C#

[Копировать](# "Копировать")

```
using System;
using System.IO;
using TFlex.DOCs.Model.FilePreview.CADService;
using TFlex.DOCs.Model.Macros;

public class ExportCadDocument : MacroProvider
{
    public ExportCadDocument(MacroContext context)
        : base(context)
    { }

    public override void Run()
    {
        // Расширение экспортируемого документа
        string extensionDoc = "pdf";
        // Путь к grb документу
        string grbPath = "Документ.grb";
        // Путь к экспортируемому документу
        string exportPath = Path.ChangeExtension("Документ", extensionDoc);

        // Вызвать функцию экспорта и получить результат о выполнении
        bool exportResult = Export(grbPath, exportPath);
    }

    // Экспортирует grb документ в другой формат
    private bool Export(string grbPath, string exportPath)
    {
        // Подключаемся к CAD-системе
        CadDocumentProvider provider = CadDocumentProvider.Connect(Context.Connection, ".grb");

        // Создаем настройки экспорта документа CAD-системы в другой формат
        ExportContext exContext = new ExportContext(exportPath);
        exContext.Page = 2; // При необходимости можем экспортировать конкретную страницу, указав ее индекс
        // exContext.Pages = new List<int>() { 1, 3, 5 }; // Можно определить список экспортируемых страниц

        // Возвращает имя файла и расширение указанной строки пути
        string fileName = Path.GetFileName(grbPath);

        // Открываем документ в режиме чтения.
        using (CadDocument document = provider.OpenDocument(grbPath, true))
        {
            // Проверяем, был ли открыт документ
            if (document is null)
            {
                Error("Ошибка экспорта", "Файл '{0}' не может быть открыт", fileName);
                return false;
            }

            // Экспортируем документ на основе заданных настроек экспорта
            // Получаем полный путь до экспортированного файла для дальнейшей проверки экспорта
            string path = document.Export(exContext);

            if (String.IsNullOrEmpty(path))
            {
                Error("Ошибка экспорта", "Файл '{0}' не может быть экспортирован", fileName);
                return false;
            }

            document.Close(false); // Закрываем документ без сохранения
        }

        return true;
    }
}
```

Для экспорта в 3D PDF необходимо установить соответствующее свойство контекста экспорта

C#

[Копировать](# "Копировать")

```
ExportContext exportContext = new ExportContext(fileName);
exportContext["PDF3D"] = true;
```

![Изображение](../icons/SectionExpanded.png)Установка размеров и разрешения изображения

C#

[Копировать](# "Копировать")

```
ExportContext exportContext = new ExportContext(fileName);
exportContext["width"] = 270; // В мм
exportContext["height"] = 150;
exportContext["resolution"] = 50;
```

![Изображение](../icons/SectionExpanded.png)Выбор алгоритма сжатия при экспорте в многостраничный TIFF

C#

[Копировать](# "Копировать")

```
ExportContext exportContext = new ExportContext(fileName);
exportContext["TiffCompression"] = TFlex.DOCs.Model.FilePreview.CADService.TiffCompressOption.Lzw.ToString(); // Указываем алгоритм сжатия
```
