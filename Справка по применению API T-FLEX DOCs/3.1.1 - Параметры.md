Команды для работы с параметрами

[Объект.Параметр](C223A4F5.htm)
[["(имя или Guid параметра)"]](618A4278.htm)
- возвращает значение параметра указанного объекта.

C#[Копировать](# "Копировать")

```
string наименование = договор.Параметр["Наименование"];
string имя = пользователь.Параметр["d6e2c40c-e3fe-401d-bf48-c9980992c47e"]; // Получаем значение параметра по его идентификатору
```

| Внимание |
| --- |
| Если наименование параметра совпадает с системным («Guid», «Имя» и т.д.), то для получения или установки значения такого параметра необходимо использовать его глобальный уникальный идентификатор (Guid) |

```
}
```

Пример обработчика события справочника «Завершение изменения параметра объекта» с помощью макроязыка.C#[Копировать](# "Копировать")

```
using System;
using System.Linq;
using TFlex.DOCs.Model.Macros;

public class Macro(MacroContext context) : MacroProvider(context)
{
    // Обработчик события "Завершение изменения параметра объекта"
    public override void Run()
    {
        // Определяем наименования параметров
        const string параметрФИО = "ФИО";
        const string параметрИмя = "Имя";
        const string параметрФамилия = "Фамилия";
        const string параметрОтчество = "Отчество";
        string[] параметрыПолногоИмени = [параметрИмя, параметрФамилия, параметрОтчество];

        // Если параметр не изменялся или изменен параметр, не относящийся к составу полного имени, выходим
        if (String.IsNullOrEmpty(ИзмененныйПараметр) || !параметрыПолногоИмени.Contains(ИзмененныйПараметр))
            return;

        // Получаем значения параметров
        string имя = Параметр[параметрИмя];
        string фамилия = Параметр[параметрФамилия];
        string отчество = Параметр[параметрОтчество];

        Параметр[параметрФИО] = $"{фамилия} {имя} {отчество}".Trim(); // Записываем полное имя в параметр
    }
}
```
