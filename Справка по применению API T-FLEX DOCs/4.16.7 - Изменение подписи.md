В данном примере рассмотрена обработка события "Изменение подписи объекта", возникающего после установки подписи

C#

```csharp
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.Signatures;

public class ChangingSignature : MacroProvider
{
    public ChangingSignature(MacroContext context)
        : base(context)
    { }

    // Обработчик события "Изменение подписи объекта"
    public override void Run()
    {
        // Получаем аргументы события "Изменение подписи объекта"
        if (Context.ModelChangedArgs is not ObjectSignatureChangedEventArgs signatureChangedEventArgs)
            return;

        Signature signature = signatureChangedEventArgs.Signature; // Подпись
        SignatureOperationType operationType = signatureChangedEventArgs.OperationType; // Тип операции, производимой с подписью
    }
}
