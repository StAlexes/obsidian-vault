Справочник представляет собой совокупность объектов определенных типов. API позволяет со стороны справочника создавать и осуществлять поиск объектов, а также предоставляет дополнительный функционал по работе со специализированными справочниками. Специализированные справочники описаны в соответствующих подразделах.

Создание экземпляра справочника

Для работы со справочником и входящими в него объектами необходимо получить описание справочника
[ReferenceInfo](D99B90CD.htm) и по нему создать экземпляр справочника.
Найти описание справочника можно по наименованию или идентификатору в каталоге справочников, доступ к которому можно получить через свойство [ReferenceCatalog](C9D72890.htm) экземпляра [ServerConnection](EAA7893.htm).

C#[Копировать](# "Копировать")

```
using System;
using TFlex.DOCs.Model;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.Structure;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        ReferenceInfo referenceInfo = Context.Connection.ReferenceCatalog.Find(Guid.Empty); // Получаем описание справочника по Guid'у
        if (referenceInfo is null || referenceInfo.ActivityStatus != GroupActivityStatus.Normal) // Если справочник не найден или деактивирован, выходим
            return;

        // Создание объекта справочника для работы с данными
        Reference reference = referenceInfo.CreateReference();

        // Загрузка объектов из справочника на первом уровне иерархии
        reference.Objects.Load();

        // Обработка загруженных объектов
        foreach (ReferenceObject referenceObject in reference.Objects)
        {
        }
    }
}
```

Создание новых объектов справочника

Для создания объекта справочника нужно вызвать метод CreateReferenceObject для инициализации объекта.
Создать объект можно без указания типа (будет использован тип по умолчанию), либо предварительно определив тип.

C#[Копировать](# "Копировать")

```
// Создание объекта с типом по умолчанию
ReferenceObject newObject1 = reference.CreateReferenceObject();
// Поиск типа
ClassObject classObject = reference.Classes.Find(/*гуид типа*/Guid.Empty);
// Создание объекта с явно заданным типом
ReferenceObject newObject2 = reference.CreateReferenceObject(classObject);
```

Затем можно приступать к редактированию созданного объекта (см. [«Редактирование объектов»](663e0b41-ed90-4985-bdca-da4741702f3e.htm#EditObjectSection)).

Создание объекта справочника по прототипу

Для создания объекта справочника по прототипу необходимо создать экземпляр справочника для работы с прототипами, найти необходимй прототип и создать контекст копирования объекта справочника

C#[Копировать](# "Копировать")

```
using System;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;

public class CreatorByPrototype : MacroProvider
{
    public CreatorByPrototype(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        var prototypeModeReference = Context.Reference.ParameterGroup.ReferenceInfo.CreateReference(true); // Создаем еще один экземпляр текущего справочника для работы с прототипами
        var prototype = prototypeModeReference.Find(1); // Находим прототип по идентификатору

        // Создаем контекст копирования объекта справочника
        var context = new CopyReferenceObjectsContext
        {
            ParentObject = Context.ReferenceObject, // В качестве родительского объекта передаем текущий
            CopyChildren = true, // Указываем, требуется ли копировать дочерние объекты
            CopyLinkedPrototypes = true // Указываем, требуется ли копировать связанные прототипы
        };

        var copySet = Context.Reference.CopyReferenceObject(prototype, context); // Создаем набор объектов на основе указанного прототипа и контекста

        var referenceObjectFromPrototype = copySet.GetNewObject(prototype); // Получаем из набора созданный по прототипу объект
        referenceObjectFromPrototype[new Guid("Guid параметра")].Value = "Новое значение парамаетра"; // При необходимости редактируем созданный объект

        bool result = copySet.EndChanges(); // Сохраняем набор объектов, созданных по прототипу
    }
}
```

Поиск объектов

Поиск объектов может осуществляться различными способами:

C#[Копировать](# "Копировать")

```
// Поиск по идентификатору объекта
int objectId = 0;
ReferenceObject changingObject = reference.Find(objectId);
// Поиск по GUID объекта
Guid objectGuid = new Guid("C9095587-24F4-4EDC-84E5-924C741A0EBA");
ReferenceObject changingObject = reference.Find(objectGuid);
```

Установка владельца для списка объектов справочникаC#[Копировать](# "Копировать")

```
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References;
using TFlex.DOCs.Model.References.Users;

public class Macro : MacroProvider
{
    public Macro(MacroContext context)
        : base(context)
    {
    }

    public override void Run()
    {
        ReferenceObject[] selectedObjects = Context.GetSelectedObjects(); // Получаем выбранные объекты
        if (Context.Reference.ParameterGroup.SupportsOwner) // Проверяем, включена ли поддержка параметра "Владелец"
        {
            // Context.Reference возвращает справочник, в контексте которого выполняется макрос.
            // Альтернативным вариантом может быть обращение к справочнику через объект: selectedObjects[0].Reference (ReferenceObject.Reference)

            var userReference = new UserReference(Context.Connection); // Создаем экземпляр справочника "Группы и пользователи"

            var groupUsers = userReference.FindOne(UserReferenceObject.Fields.FullName, "Сотрудники") as UserReferenceObject; // Находим группу пользователей "Сотрудники"
            if (groupUsers is null)
                return;

            Context.Reference.SetObjectsOwner(selectedObjects, groupUsers); // Задаем владельца
        }
    }
}
```
