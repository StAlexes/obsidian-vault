Пример создания и получения замечаний

Замечание может быть добавлено к любому объекту справочника.

C#[Копировать](# "Копировать")

```
using System.Collections.Generic;
using System.Linq;
using TFlex.DOCs.Model.Macros;
using TFlex.DOCs.Model.References.Documents;
using TFlex.DOCs.Model.References.Remarks;

public class RemarksMacro : MacroProvider
{
    public RemarksMacro(MacroContext context) : base(context)
    { }

    public override void Run()
    {
        // Находим документ, для которого получим замечания
        var documentReference = new DocumentReference(Context.Connection);
        var document = documentReference.Find(1);

        if (document is null)
            return;

        var remarkReference = new RemarksReference(Context.Connection);
        IReadOnlyCollection<RemarkReferenceObject> remarks = document.GetRemarks(); // Получаем замечания по документу
        PdfRemarkObject pdfRemark = remarks.OfType<PdfRemarkObject>().FirstOrDefault(); // Получаем pdf-замечание

        if (pdfRemark is null)
            return;

        var text = pdfRemark.Text; // Получаем текст замечания
        var status = pdfRemark.StatusType; // Получаем статус замечания

        // Добавляем текстовое замечание
        remarkReference.AddTextRemark(document, "Текст замечания");
    }
}
```
